	ErrorInfo
	cause error

func (e *Error) Unwrap() error { return e.cause }

// NewError classifies an error without serializing or replacing its native cause.
func NewError(kind ErrorKind, provider, operation, field, message string, cause error) *Error {
	if errors.Is(cause, context.Canceled) {
		kind = ErrorKindCanceled
	} else if errors.Is(cause, context.DeadlineExceeded) {
		kind = ErrorKindTimeout
	}
	return &Error{ErrorInfo: ErrorInfo{Kind: kind, Provider: provider, Operation: operation, Field: field, Message: message}, cause: cause}
}
