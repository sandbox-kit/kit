package modal

import (
	"context"

	"github.com/sandbox-kit/kit/core"
	modal "github.com/modal-labs/modal-client/go"
)

type Config struct {
	TokenID    string
	TokenSecret string
}

type Adapter struct {
	client *modal.Client
}

func New(cfg Config) (*Adapter, error) {
	client, err := modal.NewClient(
		// Illustrative: use the actual Modal SDK constructor/config.
		modal.WithToken(cfg.TokenID, cfg.TokenSecret),
	)
	if err != nil {
		return nil, err
	}

	return &Adapter{client: client}, nil
}

func (a *Adapter) CreateSandbox(
	ctx context.Context,
	opts core.SandboxOptions,
) (any, error) {
	app, err := a.client.Apps.FromName(ctx, opts.Name, nil)
	if err != nil {
		return nil, err
	}

	image := a.client.Images.FromRegistry(opts.Image, nil)

	return a.client.Sandboxes.Create(ctx, app, image, &modal.SandboxCreateParams{
		// Map only fields Modal supports.
		Region: opts.Region,
	})
}

func (a *Adapter) GetSandbox(
	ctx context.Context,
	id string,
) (any, error) {
	return a.client.Sandboxes.FromID(ctx, id, nil)
}

func (a *Adapter) DeleteSandbox(
	ctx context.Context,
	id string,
) error {
	sandbox, err := a.client.Sandboxes.FromID(ctx, id, nil)
	if err != nil {
		return err
	}

	_, err = sandbox.Terminate(ctx, nil)
	return err
}
