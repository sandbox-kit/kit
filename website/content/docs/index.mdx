Sandbox Kit is an SDK for cloud sandboxes. Install the core package and a provider, create a sandbox, then run commands and read files through the same client.

Use the language tabs and the provider control to switch snippets. Your choices stick in the browser.

<Tabs groupId="language" items={['TypeScript', 'Go']} persist>
  <Tab value="TypeScript">

<ForProvider id="daytona">

```ts
import { daytona } from "@sandbox-kit/daytona";
import { Client } from "@sandbox-kit/sandbox";

await using client = new Client({
  provider: daytona(),
  auth: { apiKey: { key: process.env.DAYTONA_API_KEY! } },
  timeout: 120_000, // default client deadline (ms)
});

const sandbox = await client.create();
console.log(sandbox.id);
```

</ForProvider>

<ForProvider id="modal">

```ts
import { modal } from "@sandbox-kit/modal";
import { Client } from "@sandbox-kit/sandbox";

await using client = new Client({
  provider: modal(),
  auth: {
    tokenPair: {
      id: process.env.MODAL_TOKEN_ID!,
      secret: process.env.MODAL_TOKEN_SECRET!,
    },
  },
  scope: {
    appName: process.env.MODAL_APP_NAME!,
    environment: process.env.MODAL_ENVIRONMENT ?? "main",
  },
  timeout: 120_000, // default client deadline (ms)
});

const sandbox = await client.create({
  source: { image: { reference: "alpine:3.21" } },
});
console.log(sandbox.id);
```

</ForProvider>

<ForProvider id="blaxel">

```ts
import { blaxel } from "@sandbox-kit/blaxel";
import { Client } from "@sandbox-kit/sandbox";

await using client = new Client({
  provider: blaxel(),
  auth: { apiKey: { key: process.env.BL_API_KEY! } },
  scope: { workspace: process.env.BL_WORKSPACE! },
  region: process.env.BL_REGION || "us-pdx-1",
  timeout: 120_000, // default client deadline (ms)
});

const sandbox = await client.create({
  name: "kit-example",
  source: { image: { reference: "blaxel/base-image:latest" } },
});
console.log(sandbox.id); // same as create name
```

</ForProvider>

  </Tab>
  <Tab value="Go">

<ForProvider id="daytona">

```go
package main

import (
	"context"
	"fmt"
	"log"
	"os"
	"time"

	"github.com/sandbox-kit/kit/sdks/go/providers/daytona"
	"github.com/sandbox-kit/kit/sdks/go/sandbox"
)

func main() {
	client, err := sandbox.NewClient(daytona.New(), &sandbox.Config{
		Auth: &sandbox.AuthConfig{Credentials: &sandbox.APIKey{
			ApiKey: &sandbox.APIKeyCredentials{Key: os.Getenv("DAYTONA_API_KEY")},
		}},
		Timeout: sandbox.Duration(2 * time.Minute),
	})
	if err != nil {
		log.Fatal(err)
	}
	defer client.Close(context.Background())

	instance, err := client.Create(context.Background(), nil)
	if err != nil {
		log.Fatal(err)
	}
	fmt.Println(instance.ID())
}
```

</ForProvider>

<ForProvider id="modal">

```go
package main

import (
	"context"
	"fmt"
	"log"
	"os"
	"time"

	"github.com/sandbox-kit/kit/sdks/go/providers/modal"
	"github.com/sandbox-kit/kit/sdks/go/sandbox"
)

func main() {
	client, err := sandbox.NewClient(modal.New(), &sandbox.Config{
		Auth: &sandbox.AuthConfig{Credentials: &sandbox.TokenPair{
			TokenPair: &sandbox.TokenPairCredentials{
				Id:     os.Getenv("MODAL_TOKEN_ID"),
				Secret: os.Getenv("MODAL_TOKEN_SECRET"),
			},
		}},
		Scope: &sandbox.Scope{
			AppName:     sandbox.Value(os.Getenv("MODAL_APP_NAME")),
			Environment: sandbox.Value("main"),
		},
		Timeout: sandbox.Duration(2 * time.Minute),
	})
	if err != nil {
		log.Fatal(err)
	}
	defer client.Close(context.Background())

	instance, err := client.Create(context.Background(), &sandbox.CreateOptions{
		Source: &sandbox.SandboxSource{Kind: &sandbox.Image{
			Image: &sandbox.ImageSource{Reference: "alpine:3.21"},
		}},
	})
	if err != nil {
		log.Fatal(err)
	}
	fmt.Println(instance.ID())
}
```

</ForProvider>

## 3. Run a command

<Tabs groupId="language" items={['TypeScript', 'Go']} persist>
  <Tab value="TypeScript">

```ts
const result = await sandbox.process.run({
  argv: ["echo", "hello from kit"],
});
// result: { exitCode, stdout, stderr }
console.log(result);
```

  </Tab>
  <Tab value="Go">

```go
import (
	"context"
	"fmt"
	"log"

	"github.com/sandbox-kit/kit/sdks/go/sandbox"
)

result, err := instance.Process().Run(context.Background(), &sandbox.ExecOptions{
	Argv: []string{"echo", "hello from kit"},
})
if err != nil {
	log.Fatal(err)
}
// result: ExitCode, Stdout, Stderr
fmt.Println(result)
```

  </Tab>
</Tabs>

A non-zero exit code is not an SDK error—check `exitCode` yourself. More options: [Run a command](/docs/process).

## 4. Write a file

<Tabs groupId="language" items={['TypeScript', 'Go']} persist>
  <Tab value="TypeScript">

```ts
await sandbox.filesystem.writeFile(
  "/tmp/note.txt",
  new TextEncoder().encode("hello\n"),
);
```

  </Tab>
  <Tab value="Go">

```go
import "context"

err = instance.Filesystem().WriteFile(
	context.Background(),
	"/tmp/note.txt",
	[]byte("hello\n"),
	nil,
)
```

  </Tab>
</Tabs>

Sandbox paths are absolute. More: [Filesystem](/docs/filesystem).

<Callout title="Close does not stop the sandbox">
  `client.close()` and handle `close` only release local resources. The cloud
  sandbox keeps running until an explicit lifecycle call or a lifetime policy
  ends it. See [Concepts](/docs/concepts) and [Lifecycle](/docs/lifecycle).
</Callout>
