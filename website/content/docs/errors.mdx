Failed calls surface a typed error with `kind`, `provider`, `operation`, `message`, and `field` when a setting was rejected.

Kinds are categories, not retry advice. Kit does not retry.

| Kind                 | What happened                                        |
| -------------------- | ---------------------------------------------------- |
| `InvalidArgument`    | A request field is invalid                           |
| `Unsupported`        | This provider does not support the call or the field |
| `Authentication`     | Credentials were rejected                            |
| `PermissionDenied`   | The account cannot perform this operation            |
| `NotFound`           | The sandbox or file path was not found               |
| `AlreadyExists`      | The named resource already exists                    |
| `Conflict`           | Current state rejects this operation                 |
| `RateLimited`        | Provider rate limit reached                          |
| `ResourceExhausted`  | Provider quota or capacity reached                   |
| `Timeout`            | The deadline fired                                   |
| `Canceled`           | The caller canceled the operation                    |
| `Unavailable`        | The provider could not be reached                    |
| `InvalidResponse`    | The provider returned an unusable response           |
| `Internal`           | Unclassified failure                                 |
| `FailedPrecondition` | Client or sandbox is not in a usable state           |
| `Unknown`            | Does not match a known category                      |

## Invalid config

Bad client config fails before any provider call.

<ForLanguage id="typescript">It throws.</ForLanguage>
<ForLanguage id="go">It returns `(*Client, error)`.</ForLanguage>

<Tabs groupId="language" items={['TypeScript', 'Go']} persist>
  <Tab value="TypeScript">

```ts
import { Client, SandboxError } from "@sandbox-kit/sandbox";

try {
  new Client({ provider, timeout: -1 });
} catch (error) {
  if (error instanceof SandboxError) {
    // error: { kind, provider, operation, field, message, rule?,
    //          statusCode?, providerCode?, providerSource?, cause? }
    console.error(error);
  }
  throw error;
}
```

  </Tab>
  <Tab value="Go">

```go
import (
	"errors"
	"fmt"
	"time"

	"github.com/sandbox-kit/kit/sdks/go/sandbox"
)

client, err := sandbox.NewClient(provider, &sandbox.Config{
	Timeout: sandbox.Duration(-1 * time.Second),
})
if err != nil {
	var detail *sandbox.Error
	if errors.As(err, &detail) {
		// detail: Kind, Provider, Operation, Field, Message, StatusCode,
		//         ProviderCode, ProviderSource, Rule (+ Unwrap cause)
		fmt.Println(detail)
	}
	return err
}
_ = client
```

  </Tab>
</Tabs>

## Handle by kind

<Tabs groupId="language" items={['TypeScript', 'Go']} persist>
  <Tab value="TypeScript">

Throws `SandboxError`.

```ts
import { ErrorKind, SandboxError } from "@sandbox-kit/sandbox";

try {
  await client.create(request);
} catch (error) {
  if (!(error instanceof SandboxError)) throw error;

  switch (error.kind) {
    case ErrorKind.InvalidArgument:
    case ErrorKind.Unsupported:
      console.error("Check configuration:", error.field, error.message);
      break;
    case ErrorKind.Authentication:
      console.error("Check your provider credentials.");
      break;
    case ErrorKind.PermissionDenied:
      console.error("Your account cannot perform this operation.");
      break;
    case ErrorKind.NotFound:
      console.error("Sandbox or file path not found.");
      break;
    case ErrorKind.AlreadyExists:
      console.error("Named resource already exists.");
      break;
    case ErrorKind.Conflict:
      console.error("Current state rejects this operation.");
      break;
    case ErrorKind.RateLimited:
    case ErrorKind.ResourceExhausted:
      console.error("Provider rate or resource limit reached.");
      break;
    case ErrorKind.Canceled:
      console.error("Operation canceled.");
      break;
    case ErrorKind.Timeout:
      console.error("Operation timed out.");
      break;
    default:
      console.error(error);
  }
}
```

The underlying cloud error is `error.cause`.

  </Tab>
  <Tab value="Go">

Returns `*sandbox.Error`.

```go
import (
	"errors"
	"fmt"

	"github.com/sandbox-kit/kit/sdks/go/sandbox"
)

var detail *sandbox.Error
if errors.As(err, &detail) {
	switch detail.Kind {
	case sandbox.ErrorKindInvalidArgument, sandbox.ErrorKindUnsupported:
		fmt.Println("Check configuration:", detail.Field)
	case sandbox.ErrorKindAuthentication:
		fmt.Println("Check your provider credentials.")
	case sandbox.ErrorKindPermissionDenied:
		fmt.Println("Your account cannot perform this operation.")
	case sandbox.ErrorKindNotFound:
		fmt.Println("Sandbox or file path not found.")
	case sandbox.ErrorKindAlreadyExists:
		fmt.Println("Named resource already exists.")
	case sandbox.ErrorKindConflict:
		fmt.Println("Current state rejects this operation.")
	case sandbox.ErrorKindRateLimited, sandbox.ErrorKindResourceExhausted:
		fmt.Println("Provider rate or resource limit reached.")
	case sandbox.ErrorKindCanceled:
		fmt.Println("Operation canceled.")
	case sandbox.ErrorKindTimeout:
		fmt.Println("Operation timed out.")
	default:
		fmt.Println(err)
	}
}
```

`Unwrap` returns the provider error, so `errors.Is(err, context.Canceled)` still matches.

</Tabs>

## Field paths

Field strings are shared (for example `resources.cpu_cores`). Rules that cover several settings use the parent (`network`).

<Tabs groupId="language" items={['TypeScript', 'Go']} persist>
  <Tab value="TypeScript">

```ts
import {
  CreateFieldResourcesCpuCores,
  SandboxError,
} from "@sandbox-kit/sandbox";

try {
  await client.create({ resources: { cpuCores: -1 } });
} catch (error) {
  if (
    error instanceof SandboxError &&
    error.field === CreateFieldResourcesCpuCores
  ) {
    console.error("bad cpuCores:", error.message);
  }
}
```

  </Tab>
  <Tab value="Go">

```go
_, err := client.Create(ctx, &sandbox.CreateOptions{
	Resources: &sandbox.Resources{CpuCores: sandbox.Value(-1.0)},
})
var detail *sandbox.Error
if errors.As(err, &detail) && detail.Field == sandbox.CreateFieldResourcesCpuCores {
	fmt.Println("bad cpuCores:", detail.Message)
}
```

  </Tab>
</Tabs>

## After failure

An error does not prove no sandbox was created—the provider may have succeeded before a later step failed, and the error does not include that id. Lifecycle failures can occur after the provider already mutated state.

<ForProvider id="daytona">

`get` before retry so you know whether the sandbox exists.

<Tabs groupId="language" items={['TypeScript', 'Go']} persist>
  <Tab value="TypeScript">

```ts
try {
  await client.create(request);
} catch (error) {
  // create may have succeeded before a later step failed — no id on the error
  if (knownId) {
    const sandbox = await client.get(knownId);
    console.log(sandbox.info());
  }
  throw error;
}
```

  </Tab>
  <Tab value="Go">

```go
instance, err := client.Create(ctx, opts)
if err != nil {
	// create may have succeeded before a later step failed — no id on the error
	if knownID != "" {
		existing, getErr := client.Get(ctx, knownID)
		if getErr == nil {
			fmt.Println(existing.Info())
		}
	}
	return err
}
_ = instance
```

  </Tab>
</Tabs>

See [Examples](/docs/examples).

</ForProvider>

<ForProvider id="blaxel">

If the name is already taken on create, you get `AlreadyExists`. If a delete is
still finishing, you get `Conflict` — wait, then `get` before you recreate that
name. Lifetime below Blaxel’s floors (≥5m max lifetime, ≥1m idle) or a
non-whole-second duration is `InvalidArgument`. Unsupported create fields
(runtime, isolation, idle disable, …) are `Unsupported`. When Blaxel returns a
control-plane code, read it on the typed error.

<ForLanguage id="typescript">Use `providerCode`.</ForLanguage>
<ForLanguage id="go">Use `ProviderCode`.</ForLanguage>

<Tabs groupId="language" items={['TypeScript', 'Go']} persist>
  <Tab value="TypeScript">

```ts
try {
  await client.create(request);
} catch (error) {
  if (!(error instanceof SandboxError)) throw error;
  if (error.kind === ErrorKind.AlreadyExists) {
    console.error("Name already taken:", error.message);
  } else if (error.kind === ErrorKind.Conflict) {
    console.error("Name still conflicting; wait and get before recreate.");
  }
  throw error;
}
```

  </Tab>
  <Tab value="Go">

```go
instance, err := client.Create(ctx, opts)
if err != nil {
	var detail *sandbox.Error
	if errors.As(err, &detail) {
		switch detail.Kind {
		case sandbox.ErrorKindAlreadyExists:
			fmt.Println("Name already taken:", detail.Message)
		case sandbox.ErrorKindConflict:
			fmt.Println("Name still conflicting; wait and Get before recreate.")
		}
	}
	return err
}
_ = instance
```

  </Tab>
</Tabs>

See [Examples](/docs/examples).

</ForProvider>
